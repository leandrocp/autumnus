## Minimal HTML

<Tabs groupId="runtime" queryString="lang" defaultValue="javascript">
  <TabItem value="javascript" label="JavaScript">

```js
import {highlight} from '@lumis-sh/lumis'
import {htmlInline} from '@lumis-sh/lumis/formatters'
import javascript from '@lumis-sh/lumis/langs/javascript'
import dracula from '@lumis-sh/themes/dracula'

const html = await highlight('const x = 1', htmlInline({language: javascript, theme: dracula}))
```

  </TabItem>
  <TabItem value="rust" label="Rust">

```rust
use lumis::{highlight, HtmlInlineBuilder, languages::Language, themes};

let html = highlight(
    "const x = 1",
    HtmlInlineBuilder::new()
        .lang(Language::Javascript)
        .theme(Some(themes::get("dracula").unwrap()))
        .build()
        .unwrap(),
);
```

  </TabItem>
  <TabItem value="elixir" label="Elixir">

```elixir
html = Lumis.highlight!("const x = 1", language: "javascript", formatter: {:html_inline, theme: "dracula"})
```

  </TabItem>
  <TabItem value="java" label="Java">

```java
import io.roastedroot.lumis4j.core.Lang;
import io.roastedroot.lumis4j.core.Lumis;
import io.roastedroot.lumis4j.core.Theme;

var lumis = Lumis.builder()
    .withLang(Lang.JAVASCRIPT)
    .withTheme(Theme.DRACULA)
    .build();

var html = lumis.highlight("const x = 1").string();
```

  </TabItem>
  <TabItem value="cli" label="CLI">

```bash
lumis highlight index.js --formatter html-inline --theme dracula
```

  </TabItem>
</Tabs>

## Multi-theme HTML

<Tabs groupId="runtime" queryString="lang" defaultValue="javascript">
  <TabItem value="javascript" label="JavaScript">

```js
import {highlight} from '@lumis-sh/lumis'
import {htmlMultiThemes} from '@lumis-sh/lumis/formatters'
import javascript from '@lumis-sh/lumis/langs/javascript'
import githubLight from '@lumis-sh/themes/github_light'
import githubDark from '@lumis-sh/themes/github_dark'

const html = await highlight(
  'const x = 1',
  htmlMultiThemes({
    language: javascript,
    themes: {light: githubLight, dark: githubDark},
    defaultTheme: 'light-dark()',
  })
)
```

  </TabItem>
  <TabItem value="rust" label="Rust">

```rust
use lumis::{highlight, HtmlMultiThemesBuilder, languages::Language, themes};
use std::collections::HashMap;

let mut theme_map = HashMap::new();
theme_map.insert("light".to_string(), themes::get("github_light").unwrap());
theme_map.insert("dark".to_string(), themes::get("github_dark").unwrap());

let html = highlight(
    "const x = 1",
    HtmlMultiThemesBuilder::new()
        .lang(Language::Javascript)
        .themes(theme_map)
        .default_theme("light-dark()")
        .build()
        .unwrap(),
);
```

  </TabItem>
  <TabItem value="elixir" label="Elixir">

```elixir
Lumis.highlight!(
  "const x = 1",
  language: "javascript",
  formatter: {:html_multi_themes,
    themes: [light: "github_light", dark: "github_dark"],
    default_theme: "light-dark()"
  }
)
```

  </TabItem>
  <TabItem value="cli" label="CLI">

```bash
lumis highlight index.js \
  --formatter html-multi-themes \
  --themes light:github_light \
  --themes dark:github_dark \
  --default-theme "light-dark()"
```

  </TabItem>
</Tabs>

## Highlight selected lines

<Tabs groupId="runtime" queryString="lang" defaultValue="javascript">
  <TabItem value="javascript" label="JavaScript">

```js
const html = await highlight(
  'line 1\nline 2\nline 3',
  htmlInline({
    language: javascript,
    theme: dracula,
    highlightLines: {lines: [2], style: 'theme'},
  })
)
```

  </TabItem>
  <TabItem value="rust" label="Rust">

```rust
use lumis::formatter::html_inline::{HighlightLines, HighlightLinesStyle};

let highlight_lines = HighlightLines {
    lines: vec![2..=2],
    style: Some(HighlightLinesStyle::Theme),
    class: None,
};

let formatter = HtmlInlineBuilder::new()
    .lang(Language::PlainText)
    .theme(Some(themes::get("dracula").unwrap()))
    .highlight_lines(Some(highlight_lines))
    .build()
    .unwrap();
```

  </TabItem>
  <TabItem value="elixir" label="Elixir">

```elixir
Lumis.highlight!(
  "line 1\nline 2\nline 3",
  language: "plaintext",
  formatter: {:html_inline,
    theme: "dracula",
    highlight_lines: %{lines: [2], style: :theme}
  }
)
```

  </TabItem>
  <TabItem value="cli" label="CLI">

```bash
lumis highlight main.rs --formatter html-inline --highlight-lines 2
```

  </TabItem>
</Tabs>

## Wrap output with custom HTML

  </TabItem>
  <TabItem value="rust" label="Rust">

```rust
use lumis::{highlight, HtmlInlineBuilder, formatter::HtmlElement, languages::Language, themes};

            open_tag: "<figure><figcaption>main.rs</figcaption>".to_string(),
            close_tag: "</figure>".to_string(),

        .build()
        .unwrap(),
);
```

## Injected languages (HTML + CSS + JS)

<Tabs groupId="runtime" queryString="lang" defaultValue="javascript">
  <TabItem value="javascript" label="JavaScript">

```js
import html from '@lumis-sh/lumis/langs/html'
import css from '@lumis-sh/lumis/langs/css'
import javascript from '@lumis-sh/lumis/langs/javascript'

// load all participating languages, not just the outer one
const hl = await createHighlighter({langs: [html, css, javascript]})
```

## Stream output to a file (Rust)

```rust
use lumis::{write_highlight, HtmlInlineBuilder, languages::Language};
use std::fs::File;

let formatter = HtmlInlineBuilder::new()
    .lang(Language::Rust)
    .build()
    .unwrap();

let mut file = File::create("output.html").unwrap();
write_highlight(&mut file, "fn main() {}", formatter).unwrap();
```

## Repeated highlighting (JavaScript)

```js
import {createHighlighter} from '@lumis-sh/lumis'
import {htmlInline} from '@lumis-sh/lumis/formatters'
import {bundledLanguages} from '@lumis-sh/lumis/bundles/web'
import dracula from '@lumis-sh/themes/dracula'

const hl = await createHighlighter({langs: [bundledLanguages]})
await hl.loadLanguage('javascript')

const html = hl.highlight('const x = 1', htmlInline({language: 'javascript', theme: dracula}))
```
