# GraphQL Schema Definition



"""

User type representing a registered user in the system

"""

type User {

  id: ID!

  email: String!

  name: String

  avatar: String

  role: Role!

  posts(first: Int, after: String): PostConnection!

  createdAt: DateTime!

  updatedAt: DateTime!

}



enum Role {

  ADMIN

  USER

  GUEST

}



type Post {

  id: ID!

  title: String!

  content: String!

  published: Boolean!

  author: User!

  tags: [Tag!]!

  comments(limit: Int = 10): [Comment!]!

  createdAt: DateTime!

}



type Tag {

  id: ID!

  name: String!

  posts: [Post!]!

}



type Comment {

  id: ID!

  text: String!

  author: User!

  post: Post!

  createdAt: DateTime!

}



# Pagination types

type PostConnection {

  edges: [PostEdge!]!

  pageInfo: PageInfo!

  totalCount: Int!

}



type PostEdge {

  cursor: String!

  node: Post!

}



type PageInfo {

  hasNextPage: Boolean!

  hasPreviousPage: Boolean!

  startCursor: String

  endCursor: String

}



# Input types

input CreatePostInput {

  title: String!

  content: String!

  published: Boolean = false

  tagIds: [ID!]

}



input UpdatePostInput {

  title: String

  content: String

  published: Boolean

  tagIds: [ID!]

}



input PostFilter {

  published: Boolean

  authorId: ID

  tagIds: [ID!]

  search: String

}



# Custom scalar

scalar DateTime



# Queries

type Query {

  me: User

  user(id: ID!): User

  users(role: Role): [User!]!



  post(id: ID!): Post

  posts(

    filter: PostFilter

    first: Int

    after: String

    orderBy: PostOrderBy

  ): PostConnection!



  tags: [Tag!]!

}



enum PostOrderBy {

  CREATED_AT_ASC

  CREATED_AT_DESC

  TITLE_ASC

  TITLE_DESC

}



# Mutations

type Mutation {

  createPost(input: CreatePostInput!): Post!

  updatePost(id: ID!, input: UpdatePostInput!): Post!

  deletePost(id: ID!): Boolean!



  publishPost(id: ID!): Post!



  addComment(postId: ID!, text: String!): Comment!

}



# Subscriptions

type Subscription {

  postCreated: Post!

  commentAdded(postId: ID!): Comment!

}
